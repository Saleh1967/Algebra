def parse_suffix(s, radical_vowel):
    """(صحيح؟، عددُ الضمائر المفعولة)."""
    L=letters(s)
    if L=="": return True,0
    best=None
    for subj in [""]+SUBJ:
        if not L.startswith(subj): continue
        rest=L[len(subj):]
        # subject «نا/ت/تم» بعد ساكن؛ و«نا» بعد متحرّك مفعول
        objs=[]; r=rest; ok=True
        while r:
            hit=next((o for o in OBJ if r.startswith(o)),None)
            if hit is None: ok=False;break
            objs.append(hit); r=r[len(hit):]
        if not ok or len(objs)>2: continue
        if subj=="" and objs and objs[0] in ("نا",) and radical_vowel=="0": continue
        if subj in ("نا","ت","تم","تما","تن","ن") and radical_vowel not in ("0",None): continue
        if objs==["ي"] and subj=="": continue   # ياءُ المخاطبة أو لام الكلمة
        cand=(len(objs), subj)
        if best is None or len(subj)>len(best[1]): best=cand
    if best is None: return False,0
    return True,best[0]
